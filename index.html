function enviarResultadosPorCorreo(nombreAlumno, modulo, calificacion, aprobado) { fetch("https://formspree.io/f/TU_CODIGO_AQUI", { method: "POST", headers: { "Content-Type": "application/json", "Accept": "application/json" }, body: JSON.stringify({ Alumno: nombreAlumno || "Sin nombre registrado", Modulo: modulo, Calificacion: calificacion + "%", Estado: aprobado ? "Aprobado" : "No aprobado", Fecha: new Date().toLocaleString() }) }) .then(response => { if (response.ok) { console.log("Resultados enviados a piedra.angulartecnologia@gmail.com"); } }) .catch(error => console.error("Error al enviar:", error)); }
